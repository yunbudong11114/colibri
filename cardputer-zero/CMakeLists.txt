cmake_minimum_required(VERSION 3.31)
project(colibri-cardputer VERSION 0.1.75 LANGUAGES C CXX)

set(COLIBRI_OFFLINE_SPEECH_PAYLOAD "" CACHE PATH "Optional ARM64 offline speech runtime and model")

option(USE_DESKTOP "Build the LVGL SDL simulator" ON)
set(COLIBRI_RUST_WORKER "" CACHE FILEPATH "Prebuilt native Rust worker (optional)")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Optimize Release artifacts without changing Debug/simulator behavior.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options($<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-ffunction-sections> $<$<CONFIG:Release>:-fdata-sections>)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_link_options($<$<CONFIG:Release>:LINKER:--gc-sections>)
    endif()
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT colibri_ipo_supported OUTPUT colibri_ipo_error)
if(colibri_ipo_supported)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()

include(FetchContent)
find_package(Freetype REQUIRED)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE FILEPATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.5.0 GIT_SHALLOW TRUE EXCLUDE_FROM_ALL)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(lvgl json)

target_compile_definitions(lvgl PUBLIC COLIBRI_DESKTOP=$<BOOL:${USE_DESKTOP}>)
target_link_libraries(lvgl PUBLIC Freetype::Freetype)
if(USE_DESKTOP)
    find_package(SDL2 CONFIG REQUIRED)
    target_link_libraries(lvgl PUBLIC SDL2::SDL2)
endif()

add_executable(colibri-cardputer
    src/main.cpp src/input.cpp src/worker_link.cpp src/markdown.cpp third_party/md4c/md4c.c)
target_include_directories(colibri-cardputer PRIVATE third_party/md4c)
set_target_properties(colibri-cardputer PROPERTIES OUTPUT_NAME colibri)
target_link_libraries(colibri-cardputer PRIVATE lvgl nlohmann_json::nlohmann_json)
if(NOT USE_DESKTOP AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(colibri-cardputer PRIVATE "LINKER:--wrap=FT_New_Face")
endif()
if(COLIBRI_RUST_WORKER)
    if(NOT EXISTS "${COLIBRI_RUST_WORKER}")
        message(FATAL_ERROR "COLIBRI_RUST_WORKER does not exist: ${COLIBRI_RUST_WORKER}")
    endif()
    set(worker_binary "${COLIBRI_RUST_WORKER}")
else()
    find_program(CARGO cargo REQUIRED)
    set(rust_source "${CMAKE_CURRENT_SOURCE_DIR}/../colibri-rust")
    set(rust_target "${CMAKE_CURRENT_BINARY_DIR}/rust")
    set(worker_binary "${rust_target}/release/colibri-cardputer-worker")
    file(GLOB_RECURSE rust_inputs CONFIGURE_DEPENDS
        "${rust_source}/src/*.rs" "${rust_source}/assets/*")
    add_custom_command(OUTPUT "${worker_binary}"
        COMMAND "${CARGO}" build --release --locked --manifest-path "${rust_source}/Cargo.toml"
                --bin colibri-cardputer-worker --target-dir "${rust_target}"
        DEPENDS ${rust_inputs} "${rust_source}/Cargo.toml" "${rust_source}/Cargo.lock"
        VERBATIM)
    add_custom_target(colibri-rust-worker ALL DEPENDS "${worker_binary}")
    add_dependencies(colibri-cardputer colibri-rust-worker)
endif()
if(USE_DESKTOP)
    target_compile_definitions(colibri-cardputer PRIVATE COLIBRI_WORKER_PATH="${worker_binary}"
        COLIBRI_EMOJI_FONT_PATH="${CMAKE_CURRENT_SOURCE_DIR}/assets/NotoEmoji.ttf")
else()
    target_compile_definitions(colibri-cardputer PRIVATE COLIBRI_WORKER_PATH="/usr/lib/colibri-cardputer/worker"
        COLIBRI_EMOJI_FONT_PATH="/usr/share/colibri-cardputer/fonts/NotoEmoji.ttf")
    install(FILES assets/NotoEmoji.ttf DESTINATION share/colibri-cardputer/fonts)
    install(FILES assets/pinyin/words.tsv DESTINATION share/colibri-cardputer/pinyin)
    install(FILES assets/pinyin/LICENSE DESTINATION share/doc/colibri-cardputer RENAME copyright-pinyin-words)
    install(FILES assets/pinyin/README.md DESTINATION share/doc/colibri-cardputer/pinyin)
    install(FILES third_party/md4c/LICENSE DESTINATION share/doc/colibri-cardputer RENAME copyright-md4c)
    install(FILES assets/LICENSE-noto-emoji.txt DESTINATION share/doc/colibri-cardputer RENAME copyright-noto-emoji)
    install(TARGETS colibri-cardputer RUNTIME DESTINATION bin)
    install(PROGRAMS "${worker_binary}" DESTINATION lib/colibri-cardputer RENAME worker)
    install(FILES assets/colibri-cardputer.png DESTINATION share/APPLaunch/share/images)
    install(FILES colibri-cardputer.desktop DESTINATION share/APPLaunch/applications)
    install(FILES "${lvgl_SOURCE_DIR}/LICENCE.txt" DESTINATION share/doc/colibri-cardputer RENAME copyright-lvgl)
    install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION share/doc/colibri-cardputer RENAME copyright-nlohmann-json)
    install(FILES "../colibri-rust/assets/LICENSE-pypinyin.txt" DESTINATION share/doc/colibri-cardputer RENAME copyright-pypinyin)
    if(COLIBRI_OFFLINE_SPEECH_PAYLOAD)
        include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/offline-speech.cmake")
        colibri_install_offline_speech("${COLIBRI_OFFLINE_SPEECH_PAYLOAD}")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(CPACK_GENERATOR DEB)
        set(CPACK_DEBIAN_COMPRESSION_TYPE xz)
        set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
        set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
        set(CPACK_DEBIAN_PACKAGE_RELEASE m5stack1)
        set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Colibri")
        set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6, libgcc-s1, libfreetype6, libstdc++6, libgomp1, fonts-noto-cjk, alsa-utils, pulseaudio-utils")
        set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Colibri chat and REPL for Cardputer Zero")
        set(CPACK_PACKAGE_FILE_NAME "Colibri_${PROJECT_VERSION}_m5stack1_arm64")
        include(CPack)
    endif()
endif()
